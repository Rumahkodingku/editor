---
title: Custom Extension
description: Add a consumer-defined Tiptap extension.
---

```tsx
import { Editor } from "@rumahkodingku/editor-react";
import { Mark } from "@tiptap/core";

const Highlight = Mark.create({
	name: "highlight",
	parseHTML: () => [{ tag: "mark" }],
	renderHTML: () => ["mark", { "data-rk-highlight": "true" }, 0],
});

const extensions = [Highlight];

export function App() {
	return <Editor extensions={extensions} defaultValue={content} />;
}
```
